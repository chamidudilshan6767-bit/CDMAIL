// විචල්‍යයන් (Variables) let currentEmail = ""; let emailLogin = ""; let emailDomain = ""; // 1. අලුත් ඊමේල් ලිපිනයක් සකස් කිරීම (Generate Email) async function generateNewEmail() { try { const response = await fetch('https://www.1secmail.com/api/v1/?action=genRandomMailbox&count=1'); const data = await response.json(); currentEmail = data[0]; // උදාහරණයක් ලෙස: abc123xyz@1secmail.com const parts = currentEmail.split('@'); emailLogin = parts[0]; emailDomain = parts[1]; // UI එකේ ඊමේල් ලිපිනය පෙන්වීම සඳහා (ඔබේ HTML එකේ id එක 'email-address' නම්) document.getElementById('email-address').value = currentEmail; // ඉන්බොක්ස් එක හිස් කිරීම document.getElementById('inbox-list').innerHTML = '

Inbox is empty. Waiting for emails...

'; } catch (error) { console.error("Error generating email:", error); } } // 2. ඉන්බොක්ස් එක පරීක්ෂා කිරීම (Check Inbox / Refresh) async function checkInbox() { if (!emailLogin || !emailDomain) return; try { const response = await fetch(`https://www.1secmail.com/api/v1/?action=getMessages&login=${emailLogin}&domain=${emailDomain}`); const messages = await response.json(); const inboxContainer = document.getElementById('inbox-list'); inboxContainer.innerHTML = ""; if (messages.length === 0) { inboxContainer.innerHTML = '

0 messages found.

'; return; } messages.forEach(msg => { const msgItem = document.createElement('div'); msgItem.style.cssText = "background: #1e1e1e; padding: 10px; margin-bottom: 8px; border-radius: 6px; cursor: pointer; color: #fff;"; msgItem.innerHTML = `From: ${msg.from}
Subject: ${msg.subject}`; // පණිවිඩය උඩ ක්ලික් කළ විට එය විවෘත කර බැලීමට msgItem.onclick = () => readMessage(msg.id); inboxContainer.appendChild(msgItem); }); } (error) { console.error("Error fetching inbox:", error); } } // 3. නිශ්චිත ඊමේල් පණිවිඩයක් කියවීම (Read Message Content) async function readMessage(id) { try { const response = await fetch(`https://www.1secmail.com/api/v1/?action=readMessage&login=${emailLogin}&domain=${emailDomain}&id=${id}`); const msgData = await response.json(); // පණිවිඩයේ සම්පූර්ණ අන්තර්ගතය පෙන්වීම alert(`From: ${msgData.from}\nSubject: ${msgData.subject}\n\n${msgData.textBody}`); } catch (error) { console.error("Error reading message:", error); } } // වෙබ් අඩුව විවෘත වූ වහාම ඊමේල් ලිපිනයක් සකස් කර ගැනීම සහ තත්පර 10කට වරක් ඉන්බොක්ස් එක Refresh වීම window.onload = () => { generateNewEmail(); setInterval(checkInbox, 10000); // 10 seconds };
